cmake_minimum_required(VERSION 3.15)

project(
    CppDevOpsDemo
    VERSION 1.0.0
    LANGUAGES CXX
)

# =============================================================
# C++ CONFIGURATION
# =============================================================

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)


# =============================================================
# APPLICATION
# =============================================================

add_executable(app
    src/main.cpp
)


# =============================================================
# INSTALLATION
# =============================================================

include(GNUInstallDirs)

install(
    TARGETS app
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)


# =============================================================
# CPACK GENERAL
# =============================================================

set(CPACK_PACKAGE_NAME "CppDevOpsDemo")

set(CPACK_PACKAGE_VENDOR
    "DevOps Demo"
)

set(CPACK_PACKAGE_DESCRIPTION_SUMMARY
    "Automated Cross-Platform C++ Application"
)

set(CPACK_PACKAGE_DESCRIPTION
    "CppDevOpsDemo cross-platform C++ application."
)

set(CPACK_PACKAGE_VERSION
    "${PROJECT_VERSION}"
)

set(CPACK_PACKAGE_CONTACT
    "developer@example.com"
)


# =============================================================
# DEBIAN PACKAGE
# =============================================================

set(CPACK_DEBIAN_PACKAGE_NAME
    "cppdevopsdemo"
)

set(CPACK_DEBIAN_PACKAGE_MAINTAINER
    "Developer"
)

set(CPACK_DEBIAN_PACKAGE_SECTION
    "utils"
)

set(CPACK_DEBIAN_PACKAGE_PRIORITY
    "optional"
)

set(CPACK_DEBIAN_PACKAGE_DESCRIPTION
    "Automated Cross-Platform C++ Application"
)


# Debian architecture
#
# x86_64 -> amd64
# ARM64  -> arm64
#

if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE "amd64")

elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
    set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE "arm64")

else()
    set(
        CPACK_DEBIAN_PACKAGE_ARCHITECTURE
        "${CMAKE_SYSTEM_PROCESSOR}"
    )
endif()


# =============================================================
# RPM PACKAGE
# =============================================================

set(CPACK_RPM_PACKAGE_NAME
    "cppdevopsdemo"
)

set(CPACK_RPM_PACKAGE_RELEASE
    "1"
)

set(CPACK_RPM_PACKAGE_LICENSE
    "MIT"
)

set(CPACK_RPM_PACKAGE_GROUP
    "Applications/Utilities"
)

set(CPACK_RPM_PACKAGE_DESCRIPTION
    "Automated Cross-Platform C++ Application"
)


# RPM architecture
#
# x86_64 -> x86_64
# ARM64  -> aarch64
#

if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    set(CPACK_RPM_PACKAGE_ARCHITECTURE "x86_64")

elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
    set(CPACK_RPM_PACKAGE_ARCHITECTURE "aarch64")

else()
    set(
        CPACK_RPM_PACKAGE_ARCHITECTURE
        "${CMAKE_SYSTEM_PROCESSOR}"
    )
endif()


# =============================================================
# WINDOWS MSI / WIX
# =============================================================

# Use WiX 3
set(CPACK_WIX_VERSION "3")

# IMPORTANT:
# Keep this GUID constant for the lifetime of the project.
set(
    CPACK_WIX_UPGRADE_GUID
    "7D5E6F3A-2D94-4F7B-8C4A-9E8D7B6A5C21"
)

# Start menu shortcut
set(
    CPACK_START_MENU_SHORTCUTS
    CppDevOpsDemo
)

# Desktop shortcut
set(
    CPACK_DESKTOP_SHORTCUTS
    CppDevOpsDemo
)


# =============================================================
# CPACK
# =============================================================

# AppImage is NOT generated by CPack.
# It is generated separately using linuxdeploy in GitHub Actions.
#
# DEB and RPM are generated explicitly by:
#
#   cpack -G DEB
#   cpack -G RPM
#
# Windows MSI is generated by:
#
#   cpack -G WIX
#

set(
    CPACK_GENERATOR
    "DEB;RPM"
)

include(CPack)
